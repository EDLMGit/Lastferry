const hash = (value: Buffer | string) => createHash('sha256').update(value).digest('hex');
function source() {
  const walk = (path: string): string[] =>
    readdirSync(path, { withFileTypes: true }).flatMap((e) =>
      e.isDirectory() ? walk(`${path}/${e.name}`) : [`${path}/${e.name}`],
    );
  const files = Object.fromEntries(
    [
      ...walk('src'),
      ...walk('data'),
      ...(existsSync('public') ? walk('public') : []),
      'package.json',
      'package-lock.json',
      'index.html',
      'vite.config.ts',
    ]
      .sort()
      .map((path) => [path, hash(readFileSync(path))]),
  );
  return {
    capturedAt: new Date().toISOString(),
    head: execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(),
    status: execFileSync('git', ['status', '--short'], { encoding: 'utf8' }).trim(),
    files,
    sha256: hash(JSON.stringify(files)),
    node: process.version,
  };
}

async function recordRuntime(page: Page, name: string) {
  const directory = `${reviewDir}/raw/${name}`;
  mkdirSync(directory, { recursive: true });
  await passive(page, () => {
    const startAudio = () => {
      const stream = (window as any).__LAST_FERRY__.captureAudioStream();
      if (!stream || (window as any).__storyRecording) return;
      const chunks: Blob[] = [],
        recorder = new MediaRecorder(stream, { mimeType: 'audio/webm;codecs=opus' });
      const start = performance.timeOrigin + performance.now();
      const stopped = new Promise<number[]>((resolve) => {
        recorder.ondataavailable = (e) => chunks.push(e.data);
        recorder.onstop = async () =>
          resolve(Array.from(new Uint8Array(await new Blob(chunks).arrayBuffer())));
      });
      (window as any).__storyRecording = { recorder, stopped, start };
      recorder.start();
    };
    document.addEventListener('click', startAudio, { capture: true, once: true });
  });
  const cdp = await page.context().newCDPSession(page),
    frames: { path: string; timestamp: number; receipt: number }[] = [];
  cdp.on('Page.screencastFrame', (event) => {
    const path = `${directory}/frame-${String(frames.length).padStart(6, '0')}.jpg`;
    writeFileSync(path, Buffer.from(event.data, 'base64'));
    frames.push({ path, timestamp: event.metadata.timestamp!, receipt: Date.now() });
    void cdp.send('Page.screencastFrameAck', { sessionId: event.sessionId });
  });
  await cdp.send('Page.startScreencast', {
    format: 'jpeg',
    quality: 87,
    maxWidth: page.viewportSize()!.width,
    maxHeight: page.viewportSize()!.height,
    everyNthFrame: 2,
  });
  return async () => {
    await cdp.send('Page.stopScreencast');
    const audioBytes = await page.evaluate(async () => {
      const capture = (window as any).__storyRecording;
      capture.recorder.stop();
      return capture.stopped as Promise<number[]>;
    });
    writeFileSync(`${directory}/runtime-audio.webm`, new Uint8Array(audioBytes));
    const manifest = {
      name,
      viewport: page.viewportSize(),
      audioStartEpochMs: await passive(page, () => (window as any).__storyRecording.start),
      audio: `${directory}/runtime-audio.webm`,
      frames,
      method:
        'Uninterrupted native viewport CDP screencast plus existing Sound.captureStream master output; timestamp alignment only, no level adjustment, normalization, substituted audio or gameplay state injection.',
    };
    save(`${name}-recording`, manifest);
    await cdp.detach();
    return manifest;
  };
}

test('first pointer and natural keyboard Play produce score output; mute and fresh final intent survive autoplay', async ({
  browser,
  baseURL,
}) => {
  const results: any[] = [];
  for (const input of ['pointer', 'keyboard', 'muted'] as const) {
    const context = await browser.newContext({ baseURL, viewport: { width: 390, height: 844 } });
    if (input === 'muted')
      await context.addInitScript(() => localStorage.setItem('last-ferry-muted', 'true'));

    expect((await audio(page)).contextState).toBe('not-created');

      await tabTo(page, 'home-play');
      await page.keyboard.press('Enter');

    await page.waitForTimeout(2400);
    const observation = await audio(page);
    results.push({ input, observation });
    if (input === 'muted') {
      expect(observation.muted).toBe(true);
      expect(observation.contextState).toBe('not-created');
      await activate(page, 'story-sound');
      await expect
        .poll(async () => (await audio(page)).scoreEffectiveOutputRms)
        .toBeGreaterThan(0.001);
      await activate(page, 'story-sound');
      expect((await audio(page)).muted).toBe(true);
    } else {
      expect(observation.scoreEffectiveOutputRms).toBeGreaterThan(0.001);
      expect(observation.contextCount).toBe(1);
      expect(observation.schedulerCount).toBe(1);
    }
    if (input === 'pointer') {
      const pending = await checkpoint(page);
      await expect.poll(async () => (await timeline(page)).shot, { timeout: 9000 }).toBe('lantern');

      await page.keyboard.down('Space');
      await expect.poll(async () => (await timeline(page)).shot).toBe('ferry');
      await page.keyboard.up('Space');
      expect(await checkpoint(page)).toBe(pending);
      await page.keyboard.press('Enter');
      expect((await state(page)).turn).toBe(0);
    }
    await context.close();

test('pending sound Retry and late completion after mute or hide use one owner', async ({
  browser,
  baseURL,
}) => {
  const records: any[] = [];
  for (const action of ['retry', 'mute', 'hide'] as const) {
    const context = await browser.newContext({
      baseURL,
      viewport: { width: 390, height: 844 },
      hasTouch: true,
    });
    await context.addInitScript(() => {
      const Native = window.AudioContext;
      const nativeState = Object.getOwnPropertyDescriptor(
        BaseAudioContext.prototype,
        'state',
      )!.get!;
      const Controlled = function () {
        const context = new Native(),
          resume = context.resume.bind(context);
        let blocked = true,
          permitted = false,
          requests = 0;
        const pending: (() => void)[] = [];
        Object.defineProperty(context, 'state', {
          configurable: true,
          get: () => (blocked ? 'suspended' : nativeState.call(context)),
        });
        context.resume = () => {
          requests++;
          return permitted
            ? ((blocked = false), resume())
            : new Promise<void>((done) => pending.push(done));
        };
        (window as any).__flowGate = {
          permit: () => {
            permitted = true;
          },
          resolve: async () => {
            blocked = false;
            await resume();
            pending.splice(0).forEach((done) => done());
          },
          status: () => ({ requests, blocked, native: nativeState.call(context) }),
        };
        return context;
      };
      Controlled.prototype = Native.prototype;
      Object.setPrototypeOf(Controlled, Native);
      Object.defineProperty(window, 'AudioContext', { configurable: true, value: Controlled });
    });
    const page = await context.newPage();
    await root(page);
    await coldTouch(page, 'home-play');
    await expect(page.locator('#story-sound')).toHaveAccessibleName('Retry sound');
    expect(await passive(page, () => (window as any).__flowGate.status().requests)).toBe(1);
    await page.waitForTimeout(2100);
    const pending = await audio(page);
    expect(pending.storyPlayed).toBe(0);
    if (action === 'retry') {
      await passive(page, () => (window as any).__flowGate.permit());
      await coldTouch(page, 'story-sound');
      await expect
        .poll(async () => (await audio(page)).scoreEffectiveOutputRms)
        .toBeGreaterThan(0.001);
      await passive(page, async () => (window as any).__flowGate.resolve());
      expect((await audio(page)).contextCount).toBe(1);
    } else {
      if (action === 'mute') {
        await more(page, 'story-settings');
        await activate(page, 'settings-mute');
      } else
        await passive(page, () => {
          Object.defineProperty(document, 'hidden', { configurable: true, value: true });
          document.dispatchEvent(new Event('visibilitychange'));
        });
      await passive(page, async () => (window as any).__flowGate.resolve());
      await page.waitForTimeout(200);
      const late = await audio(page);
      expect(late.running).toBe(false);
      expect(late.storySources).toBe(0);
      expect(late.schedulerCount).toBe(0);
      if (action === 'hide') {
        await passive(page, () => {
          Object.defineProperty(document, 'hidden', { configurable: true, value: false });
          document.dispatchEvent(new Event('visibilitychange'));
        });
        expect((await timeline(page)).paused).toBe(true);
      }
    }
    records.push({
      action,
      pending,
      after: await inspect(page),
      gate: await passive(page, () => (window as any).__flowGate.status()),
    });
    await context.close();
  }
  save('controlled-audio-intent', {
    method:
      'Real native AudioContext behind a controlled pending resume/state gate; hide is a controlled visibility notification, not physical Android evidence.',
    records,
  });
});

async function previewFit(page: Page, name: string) {
  const measure = () =>
    page.evaluate(() => {
      const failures: string[] = [];
      for (const e of document.querySelectorAll<HTMLElement>(
        '#preview .cove-choice-name, #preview .cove-choice-capacity, #preview .requirement, #preview .consequence, #commit .cove-label',
      )) {
        const range = document.createRange();
        range.selectNodeContents(e);
        for (const r of range.getClientRects()) {
          if (r.left < 0 || r.top < 0 || r.right > innerWidth + 0.5 || r.bottom > innerHeight + 0.5)
            failures.push(`${e.textContent}: viewport`);
          for (let p: HTMLElement | null = e; p; p = p.parentElement) {
            const b = p.getBoundingClientRect(),
              s = getComputedStyle(p);
            if (
              /auto|scroll|clip|hidden/.test(s.overflowX) &&
              (r.left < b.left - 0.5 || r.right > b.right + 0.5)
            )
              failures.push(`${e.textContent}: horizontal ${p.id}`);
            if (
              /auto|scroll|clip|hidden/.test(s.overflowY) &&
              (r.top < b.top - 0.5 || r.bottom > b.bottom + 0.5)
            )
              failures.push(`${e.textContent}: vertical ${p.id}`);
          }
        }
      }
      return failures;
    });
  const initial = await measure();
  await expect.poll(measure).toEqual([]);
  save(name, {
    failures: await measure(),
    initialAfterResize: initial,
    viewport: page.viewportSize(),
  });
}
